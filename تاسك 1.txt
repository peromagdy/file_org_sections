
#include <iostream>
#include <fstream>
#include <string>
using namespace std;

int main()
{
    string name;
    int id;
    float grade;

    cout << "Enter Full Name: ";
    getline(cin, name);

    cout << "Enter ID: ";
    cin >> id;

    cout << "Enter Grade: ";
    cin >> grade;

    ofstream myfile("Student.txt");

    myfile << "Student Name: " << name << endl;
    myfile << "Student ID: " << id << endl;
    myfile << "Student Grade: " << grade << endl;

    myfile.close();

    cout << "Student data saved successfully";

    return 0;
}
